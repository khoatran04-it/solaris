using System;
using System.Collections.Concurrent;
using System.Diagnostics;
using System.Linq;
using System.Net.Http;
using System.Threading.Tasks;

class Program
{
    static async Task Main(string[] args)
    {
        var targetUrl = "http://localhost:7070/api/evaluation/stress-reserve?quantity=1";
        var resetUrl = "http://localhost:7070/api/evaluation/reset-stress-inventory";
        int totalRequests = 500;

        using var handler = new SocketsHttpHandler
        {
            MaxConnectionsPerServer = 500,
            PooledConnectionLifetime = TimeSpan.FromMinutes(5)
        };
        using var client = new HttpClient(handler)
        {
            Timeout = TimeSpan.FromSeconds(30)
        };

        // 1. Reset Inventory
        Console.WriteLine("--> Resetting SKU-STRESS-50 to 50 available units...");
        var resetRes = await client.PostAsync(resetUrl, null);
        Console.WriteLine($"--> Reset Status: {resetRes.StatusCode}");

        Console.WriteLine($"--> Launching {totalRequests} concurrent requests against {targetUrl}...");

        var latencies = new ConcurrentBag<long>();
        var statusCodes = new ConcurrentBag<int>();

        var sw = Stopwatch.StartNew();
        var tasks = Enumerable.Range(0, totalRequests).Select(async i =>
        {
            var reqSw = Stopwatch.StartNew();
            try
            {
                var res = await client.PostAsync(targetUrl, null);
                reqSw.Stop();
                latencies.Add(reqSw.ElapsedMilliseconds);
                statusCodes.Add((int)res.StatusCode);
            }
            catch (Exception ex)
            {
                reqSw.Stop();
                latencies.Add(reqSw.ElapsedMilliseconds);
                statusCodes.Add(599);
            }
        });

        await Task.WhenAll(tasks);
        sw.Stop();

        int successCount = statusCodes.Count(c => c == 200);
        int outOfStockCount = statusCodes.Count(c => c == 400);
        int otherErrors = statusCodes.Count(c => c != 200 && c != 400);

        double avgLatency = latencies.Any() ? latencies.Average() : 0;
        long minLatency = latencies.Any() ? latencies.Min() : 0;
        long maxLatency = latencies.Any() ? latencies.Max() : 0;
        double throughput = totalRequests / sw.Elapsed.TotalSeconds;
        double errorRate = ((double)(totalRequests - successCount) / totalRequests) * 100.0;

        Console.WriteLine("\n==========================================================================");
        Console.WriteLine("               SOLARIS CONCURRENCY STRESS TEST BENCHMARK REPORT           ");
        Console.WriteLine("==========================================================================");
        Console.WriteLine($"Target SKU               : SKU-STRESS-50 (Thịt Ba Chỉ Heo)");
        Console.WriteLine($"Initial Available Stock  : 50.000");
        Console.WriteLine($"Concurrent Users (CCU)   : {totalRequests}");
        Console.WriteLine($"Total Execution Time     : {sw.ElapsedMilliseconds} ms ({sw.Elapsed.TotalSeconds:F2} s)");
        Console.WriteLine("--------------------------------------------------------------------------");
        Console.WriteLine($"Successful Orders (200 OK)       : {successCount} (Reserved 50 units)");
        Console.WriteLine($"Rejected Orders (400 Out of Stock): {outOfStockCount}");
        Console.WriteLine($"Unexpected Errors                : {otherErrors}");
        Console.WriteLine($"Error Percentage (Expected)      : {errorRate:F2}%");
        Console.WriteLine("--------------------------------------------------------------------------");
        Console.WriteLine($"Average Response Time    : {avgLatency:F2} ms");
        Console.WriteLine($"Min Response Time        : {minLatency} ms");
        Console.WriteLine($"Max Response Time        : {maxLatency} ms");
        Console.WriteLine($"Throughput               : {throughput:F2} req/sec");
        Console.WriteLine("==========================================================================");
    }
}
